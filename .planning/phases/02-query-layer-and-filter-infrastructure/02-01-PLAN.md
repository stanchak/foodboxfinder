   - **modelType**: Validate `searchParams.model` against MODEL_TYPE_GROUPS keys. Store the matched group's `matchPattern` value.
   - **geography**: Validate `searchParams.geo` against GEOGRAPHY_GROUPS keys. Store the matched group's `matchPattern` value.
   - **status**: Use `asArray(searchParams.status)`, validate each is a valid ProviderStatus. Default: `["ACTIVE", "HYBRID"]`.
   - **sortBy**: Validate `searchParams.sort` against SORT_OPTIONS. Default: `"featured"`.
   - **page**: Parse `searchParams.page` as integer, clamp to >= 1. Default: 1.
   - **pageSize**: Parse `searchParams.pageSize` as integer, clamp to 1-48. Default: 12.
   - All invalid values silently fall back to defaults -- never throw.

   ```typescript
   export function nullAwareStringFilter(field: string, value: string | undefined): Prisma.ProviderWhereInput | Record<string, never> {
     if (!value) return {};
     return {
       OR: [
         { [field]: { contains: value, mode: "insensitive" as const } },
         { [field]: null },
         { [field]: "" },
       ],
     };
   }

   export function nullAwareEnumFilter(field: string, value: string | undefined): Prisma.ProviderWhereInput | Record<string, never> {
     if (!value) return {};
     return {
       OR: [
         { [field]: value },
         { [field]: null },
       ],
     };
   }
   ```
