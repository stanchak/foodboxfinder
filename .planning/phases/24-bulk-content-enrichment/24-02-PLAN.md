```bash

import 'dotenv/config';
import { PrismaClient } from './src/generated/prisma/client';
import { PrismaPg } from '@prisma/adapter-pg';
const prisma = new PrismaClient({ adapter: new PrismaPg({ connectionString: process.env.DATABASE_URL! }) });
async function check() {
  // Check for any descriptions that still look like stubs
  const suspiciousDesc = await prisma.provider.findMany({
    where: { status: { not: 'DISCONTINUED' }, OR: [

  // Check shortDescription length constraint

    where: { status: { not: 'DISCONTINUED' }, shortDescription: { not: null } },

  const tooLong = longShort.filter(p => p.shortDescription && p.shortDescription.length > 300);
  console.log('\\nShortDescriptions over 300 chars:', tooLong.length);
  tooLong.forEach(p => console.log('  ', p.slug, p.shortDescription?.length));

  // Check valueTier distribution
  const tiers = await prisma.provider.groupBy({
    by: ['valueTier'],
    where: { status: { not: 'DISCONTINUED' } },

  });
  console.log('\\nValueTier distribution:');
  tiers.forEach(t => console.log('  ', t.valueTier, ':', t._count));

  // Check dietary tag distribution
  const tagCounts = await prisma.providerDietaryTag.groupBy({
    by: ['tag'],

  await prisma.\$disconnect();
}
check();
```
